local function refreshWhitelistGUI()

	for _, child in ipairs(scrolling:GetChildren()) do
		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	-- Add buttons for each player

		if player ~= LocalPlayer then
			local btn = Instance.new("TextButton")
			btn.Size = UDim2.new(1, -10, 0, 26)
			btn.BackgroundColor3 = whitelist[player] and Color3.fromRGB(0, 120, 60) or Color3.fromRGB(40, 40, 40)
			btn.BackgroundTransparency = 0.2
			btn.BorderSizePixel = 0
			btn.Text = player.Name
			btn.TextColor3 = Color3.fromRGB(255, 255, 255)
			btn.TextSize = 13
			btn.Font = Enum.Font.Gotham
			btn.TextTruncate = Enum.TextTruncate.AtEnd

			btnCorner.CornerRadius = UDim.new(0, 6)
			btnCorner.Parent = btn

			btn.MouseButton1Click:Connect(function()
				if whitelist[player] then
					whitelist[player] = nil

				else
					whitelist[player] = true
					btn.BackgroundColor3 = Color3.fromRGB(0, 120, 60)
				end
			end)
		end
	end
end

-- Check if player has force field
local function hasForceField(character)
	if not character then return false end
	
	-- Check for ForceField instance
	for _, child in ipairs(character:GetChildren()) do
		if child:IsA("ForceField") then
			return true
		end
	end
	
	-- Check for common force field names in character or parts
	local forceFieldNames = {"ForceField", "SpawnShield", "Shield", "InvincibilityShield"}
	
	for _, part in ipairs(character:GetChildren()) do
		if part:IsA("BasePart") then
			for _, child in ipairs(part:GetChildren()) do
				if child:IsA("ForceField") then
					return true
				end
			end
		end
	end
	
	return false
end

-- Check if player is alive
local function isPlayerAlive(player)
	if not player then return false end
	
	local character = player.Character
	if not character then return false end
	
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return false end
	
	-- Check if health is above 0 and humanoid isn't dead
	if humanoid.Health <= 0 then return false end
	
	-- Check if humanoid state is dead
	if humanoid:GetState() == Enum.HumanoidStateType.Dead then return false end
	
	return true
end

local function hasWallBetween(fromPos, toPos)
	local direction = toPos - fromPos
	local distance = direction.Magnitude
	if distance < 1 then return false end

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {LocalPlayer.Character}

	local result = workspace:Raycast(fromPos, direction.Unit * distance, params)

	if result then
		local hitChar = result.Instance:FindFirstAncestorOfClass("Model")
		if hitChar and Players:GetPlayerFromCharacter(hitChar) then
			return false
		end
		return true
	end
	return false
end

local function getClosestPlayer()
	local character = LocalPlayer.Character
	if not character then return nil end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	if not myHRP then return nil end

	local closestPlayer = nil
	local shortestDistance = math.huge

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character and not whitelist[player] then
			-- Skip dead players
			if not isPlayerAlive(player) then continue end
			
			-- Skip players with force field
			if hasForceField(player.Character) then continue end
			
			local hrp = player.Character:FindFirstChild("HumanoidRootPart")
			local humanoid = player.Character:FindFirstChildOfClass("Humanoid")

			if hrp and humanoid and humanoid.Health > 0 then
				local distance = (myHRP.Position - hrp.Position).Magnitude

				if distance <= MAX_RANGE and distance < shortestDistance then
					if not hasWallBetween(myHRP.Position, hrp.Position) then
						shortestDistance = distance
						closestPlayer = player
					end
				end
			end
		end
	end

	return closestPlayer
end

local function tryReload(tool)
	if tool.Name ~= "Nambu" then return end

	local reloadEvent = tool:FindFirstChild("ReloadEvent")
		or tool:FindFirstChild("Reload")
		or tool:FindFirstChild("RemoteReload")
		or tool:FindFirstChild("ReloadRemote")

	if reloadEvent and reloadEvent:IsA("RemoteEvent") then
		reloadEvent:FireServer()
	end
end

-- Main loop
while true do
	task.wait(0.03)

	local character = LocalPlayer.Character
	if not character then continue end

	local tool = character:FindFirstChild("Howa") or character:FindFirstChild("Nambu")
	if not tool then continue end

	local fireEvent = tool:FindFirstChild("FireEvent")
	if not fireEvent then continue end

	local myHRP = character:FindFirstChild("HumanoidRootPart")
	if not myHRP then continue end

	local originPart = tool:FindFirstChild("Handle") or myHRP
	local base = originPart.Position

	currentTarget = getClosestPlayer()
	updateGUI()

	if not currentTarget or not currentTarget.Character then continue end

	-- Double check if target is still valid (alive and no force field)
	if not isPlayerAlive(currentTarget) or hasForceField(currentTarget.Character) then
		currentTarget = nil
		updateGUI()
		continue
	end

	local humanoid = currentTarget.Character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then
		currentTarget = nil
		updateGUI()
		continue
	end

	local targetPart
	if humanoid.Health <= 15 then
		targetPart = currentTarget.Character:FindFirstChild("UpperTorso")
			or currentTarget.Character:FindFirstChild("Torso")
	else
		targetPart = currentTarget.Character:FindFirstChild("HumanoidRootPart")
	end

	if not targetPart then continue end

	local args = {
		vector.create(base.X, base.Y, base.Z),
		targetPart,
		vector.create(base.X + 0.42, base.Y - 1, base.Z + 0.52)
	}

	fireEvent:FireServer(unpack(args))
	tryReload(tool)
end
